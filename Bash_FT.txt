#!/bin/bash
set -e  # stop the script if any command fails

# -------------------------------
# Step 1: Install dependencies (only once)
# -------------------------------
echo ">>> Installing dependencies..."
pip install diffusers transformers accelerate safetensors torch

# -------------------------------
# Step 2: Set the base model
# -------------------------------
export MODEL_NAME="stable-diffusion-v1-5/stable-diffusion-v1-5"

# -------------------------------
# Step 3: Run 6 fine-tunings one after the other
# -------------------------------

# Array of instance directories (your training datasets)
INSTANCE_DIRS=(
  "./dog"
  "./cat"
  "./car"
  "./flower"
  "./person1"
  "./person2"
)

# Array of corresponding output directories
OUTPUT_DIRS=(
  "saved_models/dog"
  "saved_models/cat"
  "saved_models/car"
  "saved_models/flower"
  "saved_models/person1"
  "saved_models/person2"
)

# Loop through each pair and fine-tune
for i in ${!INSTANCE_DIRS[@]}; do
  export INSTANCE_DIR="${INSTANCE_DIRS[$i]}"
  export OUTPUT_DIR="${OUTPUT_DIRS[$i]}"

  echo ">>> Starting fine-tuning run $((i+1)) with INSTANCE_DIR=$INSTANCE_DIR and OUTPUT_DIR=$OUTPUT_DIR"

  accelerate launch train_dreambooth.py \
    --pretrained_model_name_or_path=$MODEL_NAME \
    --instance_data_dir=$INSTANCE_DIR \
    --output_dir=$OUTPUT_DIR \
    --instance_prompt="a photo of sks object" \
    --resolution=512 \
    --train_batch_size=1 \
    --gradient_accumulation_steps=1 \
    --learning_rate=5e-6 \
    --lr_scheduler="constant" \
    --lr_warmup_steps=0 \
    --max_train_steps=400 \
    --push_to_hub

  echo ">>> Finished run $((i+1)), clearing GPU memory..."
  python -c "import torch; torch.cuda.empty_cache()"
  sleep 5  # short pause before next run
done

echo ">>> All 6 fine-tunings finished successfully!"
